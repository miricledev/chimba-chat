// client/src/pages/AdminStoryCreate.jsx
import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import axios from 'axios';
import { toast } from 'react-toastify';
import { API_URL } from '../config';
import { FaPlus, FaTrash, FaSave, FaArrowLeft } from 'react-icons/fa';

const AdminStoryCreate = () => {
  const [title, setTitle] = useState('');
  const [description, setDescription] = useState('');
  const [level, setLevel] = useState('beginner');
  const [tags, setTags] = useState('');
  const [isPremium, setIsPremium] = useState(false);
  const [messages, setMessages] = useState([
    { text: 'Hola! Bienvenido a nuestra historia.', order: 0, quizType: 'none', translations: {} }
  ]);
  const [loading, setLoading] = useState(false);
  const navigate = useNavigate();
  
  const handleAddMessage = () => {
    setMessages([
      ...messages,
      { 
        text: '', 
        order: messages.length, 
        quizType: 'none',
        options: [],
        correctOption: '',
        translations: {} 
      }
    ]);
  };
  
  const handleRemoveMessage = (index) => {
    const newMessages = [...messages];
    newMessages.splice(index, 1);
    
    // Update the order of remaining messages
    const updatedMessages = newMessages.map((msg, idx) => ({
      ...msg,
      order: idx
    }));
    
    setMessages(updatedMessages);
  };
  
  const handleMessageChange = (index, field, value) => {
    const newMessages = [...messages];
    newMessages[index][field] = value;
    setMessages(newMessages);
  };
  
  const handleOptionsChange = (index, options) => {
    const optionsArray = options.split(',').map(opt => opt.trim()).filter(opt => opt);
    const newMessages = [...messages];
    newMessages[index].options = optionsArray;
    setMessages(newMessages);
  };
  
  const handleTranslationsChange = (index, translationsText) => {
    try {
      // Parse translations from text format (word:translation, word:translation)
      const translationsObj = {};
      translationsText.split(',').forEach(pair => {
        const [word, translation] = pair.split(':').map(item => item.trim());
        if (word && translation) {
          translationsObj[word.toLowerCase()] = translation;
        }
      });
      
      const newMessages = [...messages];
      newMessages[index].translations = translationsObj;
      setMessages(newMessages);
    } catch (error) {
      console.error('Error parsing translations:', error);
    }
  };
  
  const handleSubmit = async (e) => {
    e.preventDefault();
    
    if (!title || !description || messages.some(msg => !msg.text)) {
      toast.error('Please fill in all required fields');
      return;
    }
    
    const storyData = {
      title,
      description,
      level,
      tags: tags.split(',').map(tag => tag.trim()).filter(tag => tag),
      isPremium,
      messages: messages.map(msg => ({
        ...msg,
        xpValue: msg.quizType !== 'none' ? 10 : 5 // More XP for quiz messages
      }))
    };
    
    try {
      setLoading(true);
      await axios.post(`${API_URL}/admin/stories`, storyData);
      toast.success('Story created successfully');
      navigate('/admin');
    } catch (error) {
      console.error('Error creating story:', error);
      toast.error('Failed to create story');
      setLoading(false);
    }
  };
  
  return (
    <div className="max-w-4xl mx-auto">
      <div className="flex items-center mb-6">
        <button 
          onClick={() => navigate('/admin')}
          className="mr-4 btn bg-white text-gray-700 hover:bg-gray-100 shadow-sm"
        >
          <FaArrowLeft className="mr-2" /> Back to Dashboard
        </button>
        
        <h1 className="text-2xl font-bold text-gray-900">Create New Story</h1>
      </div>
      
      <form onSubmit={handleSubmit} className="bg-white rounded-lg shadow-md p-6">
        <div className="grid grid-cols-1 gap-6">
          {/* Story details */}
          <div className="space-y-4">
            <h2 className="text-xl font-semibold border-b pb-2">Story Details</h2>
            
            <div>
              <label htmlFor="title" className="form-label">Title</label>
              <input
                id="title"
                type="text"
                className="form-input"
                value={title}
                onChange={(e) => setTitle(e.target.value)}
                required
              />
            </div>
            
            <div>
              <label htmlFor="description" className="form-label">Description</label>
              <textarea
                id="description"
                className="form-input h-24"
                value={description}
                onChange={(e) => setDescription(e.target.value)}
                required
              />
            </div>
            
            <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
              <div>
                <label htmlFor="level" className="form-label">Level</label>
                <select
                  id="level"
                  className="form-input"
                  value={level}
                  onChange={(e) => setLevel(e.target.value)}
                >
                  <option value="beginner">Beginner</option>
                  <option value="intermediate">Intermediate</option>
                  <option value="advanced">Advanced</option>
                </select>
              </div>
              
              <div>
                <label htmlFor="tags" className="form-label">Tags (comma separated)</label>
                <input
                  id="tags"
                  type="text"
                  className="form-input"
                  value={tags}
                  onChange={(e) => setTags(e.target.value)}
                  placeholder="travel, food, family"
                />
              </div>
              
              <div className="flex items-center mt-8">
                <input
                  id="isPremium"
                  type="checkbox"
                  className="h-4 w-4 text-indigo-600 focus:ring-indigo-500 border-gray-300 rounded"
                  checked={isPremium}
                  onChange={(e) => setIsPremium(e.target.checked)}
                />
                <label htmlFor="isPremium" className="ml-2 block text-gray-700">
                  Premium Story
                </label>
              </div>
            </div>
          </div>
          
          {/* Messages */}
          <div className="space-y-6 mt-4">
            <h2 className="text-xl font-semibold border-b pb-2">Messages</h2>
            
            {messages.map((message, index) => (
              <div key={index} className="bg-gray-50 p-4 rounded-md relative space-y-3">
                <button
                  type="button"
                  className="absolute top-2 right-2 text-red-500 hover:text-red-700"
                  onClick={() => handleRemoveMessage(index)}
                  disabled={messages.length === 1}
                >
                  <FaTrash />
                </button>
                
                <div>
                  <label className="form-label">Message {index + 1}</label>
                  <textarea
                    className="form-input h-20"
                    value={message.text}
                    onChange={(e) => handleMessageChange(index, 'text', e.target.value)}
                    required
                  />
                </div>
                
                <div>
                  <label className="form-label">Quiz Type</label>
                  <select
                    className="form-input"
                    value={message.quizType}
                    onChange={(e) => handleMessageChange(index, 'quizType', e.target.value)}
                  >
                    <option value="none">No Quiz</option>
                    <option value="multiple-choice">Multiple Choice</option>
                    <option value="fill-blank">Fill in the Blank</option>
                    <option value="word-match">Word Matching</option>
                  </select>
                </div>
                
                {message.quizType !== 'none' && (
                  <>
                    <div>
                      <label className="form-label">Quiz Question</label>
                      <input
                        type="text"
                        className="form-input"
                        value={message.quizQuestion || ''}
                        onChange={(e) => handleMessageChange(index, 'quizQuestion', e.target.value)}
                        placeholder="What does this word mean?"
                      />
                    </div>
                    
                    <div>
                      <label className="form-label">Options (comma separated)</label>
                      <input
                        type="text"
                        className="form-input"
                        value={message.options ? message.options.join(', ') : ''}
                        onChange={(e) => handleOptionsChange(index, e.target.value)}
                        placeholder="Option 1, Option 2, Option 3"
                      />
                    </div>
                    
                    <div>
                      <label className="form-label">Correct Option</label>
                      <input
                        type="text"
                        className="form-input"
                        value={message.correctOption || ''}
                        onChange={(e) => handleMessageChange(index, 'correctOption', e.target.value)}
                        placeholder="The correct answer"
                      />
                    </div>
                  </>
                )}
                
                <div>
                  <label className="form-label">
                    Word Translations (format: word:translation, word:translation)
                  </label>
                  <textarea
                    className="form-input h-16"
                    value={Object.entries(message.translations || {})
                      .map(([word, translation]) => `${word}:${translation}`)
                      .join(', ')}
                    onChange={(e) => handleTranslationsChange(index, e.target.value)}
                    placeholder="hola:hello, adiós:goodbye"
                  />
                </div>
              </div>
            ))}
            
            <button
              type="button"
              className="btn bg-gray-100 text-gray-700 hover:bg-gray-200 flex items-center"
              onClick={handleAddMessage}
            >
              <FaPlus className="mr-2" /> Add Message
            </button>
          </div>
          
          <div className="flex justify-end pt-4 border-t">
            <button
              type="button"
              className="btn btn-secondary mr-4"
              onClick={() => navigate('/admin')}
            >
              Cancel
            </button>
            
            <button
              type="submit"
              className="btn btn-primary flex items-center"
              disabled={loading}
            >
              {loading ? (
                <div className="animate-spin rounded-full h-5 w-5 border-t-2 border-b-2 border-white mr-2"></div>
              ) : (
                <FaSave className="mr-2" />
              )}
              Create Story
            </button>
          </div>
        </div>
      </form>
    </div>
  );
};

export default AdminStoryCreate;

      console.error('Error parsing translations:', error);
    }
  };

// client/src/contexts/AuthContext.jsx
import { createContext, useContext, useState, useEffect } from 'react';
import axios from 'axios';
import { API_URL } from '../config';

const AuthContext = createContext();

export const useAuth = () => useContext(AuthContext);

export const AuthProvider = ({ children }) => {
  const [currentUser, setCurrentUser] = useState(null);
  const [loading, setLoading] = useState(true);
  
  // Set up axios default headers
  const setAuthToken = (token) => {
    if (token) {
      axios.defaults.headers.common['Authorization'] = `Bearer ${token}`;
    } else {
      delete axios.defaults.headers.common['Authorization'];
    }
  };
  
  // Check if user is logged in on app load
  useEffect(() => {
    const checkLoggedIn = async () => {
      setLoading(true);
      try {
        const token = localStorage.getItem('token');
        
        if (token) {
          setAuthToken(token);
          const { data } = await axios.get(`${API_URL}/auth/me`);
          setCurrentUser(data);
        }
      } catch (error) {
        console.error('Auth error:', error);
        localStorage.removeItem('token');
        setAuthToken(null);
      } finally {
        setLoading(false);
      }
    };
    
    checkLoggedIn();
  }, []);
  
  // Login function
  const login = async (email, password) => {
    const { data } = await axios.post(`${API_URL}/auth/login`, { email, password });
    setCurrentUser(data);
    localStorage.setItem('token', data.token);
    setAuthToken(data.token);
    return data;
  };
  
  // Register function
  const register = async (username, email, password) => {
    const { data } = await axios.post(`${API_URL}/auth/register`, { 
      username, email, password 
    });
    setCurrentUser(data);
    localStorage.setItem('token', data.token);
    setAuthToken(data.token);
    return data;
  };
  
  // Logout function
  const logout = () => {
    setCurrentUser(null);
    localStorage.removeItem('token');
    setAuthToken(null);
  };
  
  // Check subscription status
  const checkSubscription = async () => {
    try {
      const { data } = await axios.get(`${API_URL}/payments/subscription-status`);
      
      // Update user premium status
      setCurrentUser(prev => ({
        ...prev,
        isPremium: data.isPremium
      }));
      
      return data.isPremium;
    } catch (error) {
      console.error('Error checking subscription:', error);
      return false;
    }
  };
  
  // Update user XP
  const updateUserXp = (xpAmount) => {
    if (currentUser) {
      setCurrentUser(prev => ({
        ...prev,
        xp: (prev.xp || 0) + xpAmount
      }));
    }
  };
  
  const value = {
    currentUser,
    loading,
    login,
    register,
    logout,
    checkSubscription,
    updateUserXp
  };
  
  return (
    <AuthContext.Provider value={value}>
      {children}
    </AuthContext.Provider>
  );
};

// client/src/components/MessageBubble.jsx
import { useState } from 'react';

const MessageBubble = ({ message, onWordClick }) => {
  const processText = (text, translations) => {
    if (!translations || Object.keys(translations).length === 0) {
      return <span>{text}</span>;
    }
    
    const words = text.split(' ');
    
    return words.map((word, index) => {
      // Clean the word for lookup (remove punctuation)
      const cleanWord = word.toLowerCase().replace(/[.,?!;:'"()]/g, '');
      
      // Check if this word has a translation
      if (translations[cleanWord]) {
        return (
          <span key={index}>
            <span 
              className="highlighted-word"
              onClick={() => onWordClick(cleanWord, translations[cleanWord])}
            >
              {word}
            </span>
            {' '}
          </span>
        );
      }
      
      return <span key={index}>{word} </span>;
    });
  };
  
  return (
    <div className={`text-bubble ${message.isSender ? 'text-bubble-sender' : 'text-bubble-receiver'}`}>
      {processText(message.text, message.translations)}
    </div>
  );
};

export default MessageBubble;

import { useState } from 'react';

const QuizComponent = ({ quiz, onAnswer }) => {
  const [selectedOption, setSelectedOption] = useState(null);
  const [answered, setAnswered] = useState(false);
  
  const handleOptionSelect = (option) => {
    if (answered) return;
    setSelectedOption(option);
  };
  
  const handleSubmit = () => {
    if (!selectedOption || answered) return;
    
    const isCorrect = selectedOption === quiz.correctOption;
    setAnswered(true);
    onAnswer(isCorrect);
  };
  
  const getOptionClass = (option) => {
    if (!answered) {
      return selectedOption === option 
        ? 'bg-indigo-100 border-indigo-500'
        : 'bg-white hover:bg-gray-50';
    }
    
    if (option === quiz.correctOption) {
      return 'bg-green-100 border-green-500';
    }
    
    if (selectedOption === option && option !== quiz.correctOption) {
      return 'bg-red-100 border-red-500';
    }
    
    return 'bg-white opacity-60';
  };
  
  // Render quiz based on type
  if (quiz.quizType === 'multiple-choice') {
    return (
      <div className="bg-white rounded-lg shadow-md p-4 my-4">
        <h3 className="font-medium text-gray-800 mb-3">{quiz.question}</h3>
        
        <div className="space-y-2">
          {quiz.options.map((option, index) => (
            <button
              key={index}
              className={`w-full text-left p-3 border rounded-md transition ${getOptionClass(option)}`}
              onClick={() => handleOptionSelect(option)}
              disabled={answered}
            >
              {option}
            </button>
          ))}
        </div>
        
        <div className="mt-4 flex justify-end">
          <button
            className="btn btn-primary"
            onClick={handleSubmit}
            disabled={!selectedOption || answered}
          >
            {answered ? 'Continue' : 'Check Answer'}
          </button>
        </div>
        
        {answered && (
          <div className={`mt-3 p-3 rounded-md ${selectedOption === quiz.correctOption ? 'bg-green-50 text-green-800' : 'bg-red-50 text-red-800'}`}>
            {selectedOption === quiz.correctOption 
              ? '¡Correcto! Great job!'
              : `Incorrect. The correct answer is: ${quiz.correctOption}`
            }
          </div>
        )}
      </div>
    );
  }
  
  return null;
};

export default QuizComponent;

// client/src/pages/Dashboard.jsx
import { useState, useEffect } from 'react';
import { Link } from 'react-router-dom';
import axios from 'axios';
import { toast } from 'react-toastify';
import { API_URL } from '../config';
import StoryCard from '../components/StoryCard';
import { FaSearch, FaFilter } from 'react-icons/fa';

const Dashboard = () => {
  const [stories, setStories] = useState([]);
  const [loading, setLoading] = useState(true);
  const [searchTerm, setSearchTerm] = useState('');
  const [level, setLevel] = useState('');
  const [tag, setTag] = useState('');
  const [availableTags, setAvailableTags] = useState([]);
  
  useEffect(() => {
    fetchStories();
  }, [level, tag]);
  
  const fetchStories = async () => {
    try {
      setLoading(true);
      const params = {};
      
      if (level) params.level = level;
      if (tag) params.tag = tag;
      
      const { data } = await axios.get(`${API_URL}/stories`, { params });
      setStories(data);
      
      // Extract unique tags from all stories
      const tags = new Set();
      data.forEach(story => {
        if (story.tags && Array.isArray(story.tags)) {
          story.tags.forEach(tag => tags.add(tag));
        }
      });
      setAvailableTags(Array.from(tags));
    } catch (error) {
      console.error('Error fetching stories:', error);
      toast.error('Failed to load stories');
    } finally {
      setLoading(false);
    }
  };
  
  const filteredStories = stories.filter(story => 
    story.title.toLowerCase().includes(searchTerm.toLowerCase()) ||
    story.description.toLowerCase().includes(searchTerm.toLowerCase())
  );
  
  return (
    <div className="container mx-auto px-4 py-8">
      <div className="flex flex-col md:flex-row justify-between items-start md:items-center mb-8">
        <h1 className="text-3xl font-bold text-gray-900 mb-4 md:mb-0">Spanish Stories</h1>
        
        <div className="w-full md:w-auto flex flex-col sm:flex-row gap-4">
          <div className="relative">
            <input
              type="text"
              placeholder="Search stories..."
              className="form-input pl-10 pr-4 py-2 w-full md:w-64"
              value={searchTerm}
              onChange={(e) => setSearchTerm(e.target.value)}
            />
            <FaSearch className="absolute left-3 top-1/2 transform -translate-y-1/2 text-gray-400" />
          </div>
          
          <div className="flex gap-2">
            <select
              className="form-input py-2"
              value={level}
              onChange={(e) => setLevel(e.target.value)}
            >
              <option value="">All Levels</option>
              <option value="beginner">Beginner</option>
              <option value="intermediate">Intermediate</option>
              <option value="advanced">Advanced</option>
            </select>
            
            <select
              className="form-input py-2"
              value={tag}
              onChange={(e) => setTag(e.target.value)}
            >
              <option value="">All Tags</option>
              {availableTags.map((tag, index) => (
                <option key={index} value={tag}>{tag}</option>
              ))}
            </select>
          </div>
        </div>
      </div>
      
      {loading ? (
        <div className="flex justify-center py-12">
          <div className="animate-spin rounded-full h-12 w-12 border-t-2 border-b-2 border-indigo-600"></div>
        </div>
      ) : filteredStories.length === 0 ? (
        <div className="bg-white rounded-lg shadow-md p-8 text-center">
          <h2 className="text-xl font-medium text-gray-700 mb-2">No stories found</h2>
          <p className="text-gray-500 mb-4">
            Try adjusting your search or filters to find more stories.
          </p>
          <button 
            onClick={() => {
              setSearchTerm('');
              setLevel('');
              setTag('');
            }}
            className="btn btn-primary"
          >
            Clear Filters
          </button>
        </div>
      ) : (
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
          {filteredStories.map(story => (
            <StoryCard key={story.id} story={story} />
          ))}
        </div>
      )}
    </div>
  );
};

export default Dashboard;

// client/src/pages/StoryView.jsx
import { useState, useEffect, useRef } from 'react';
import { useParams, useNavigate } from 'react-router-dom';
import axios from 'axios';
import { toast } from 'react-toastify';
import { API_URL } from '../config';
import { useAuth } from '../contexts/AuthContext';
import MessageBubble from '../components/MessageBubble';
import QuizComponent from '../components/QuizComponent';
import PremiumModal from '../components/PremiumModal';
import WordTooltip from '../components/WordTooltip';
import { FaArrowLeft, FaArrowRight, FaSpinner } from 'react-icons/fa';

const StoryView = () => {
  const { id } = useParams();
  const { currentUser, updateUserXp } = useAuth();
  const [story, setStory] = useState(null);
  const [loading, setLoading] = useState(true);
  const [currentMessageIndex, setCurrentMessageIndex] = useState(0);
  const [displayedMessages, setDisplayedMessages] = useState([]);
  const [userProgress, setUserProgress] = useState(null);
  const [showPremiumModal, setShowPremiumModal] = useState(false);
  const [selectedWord, setSelectedWord] = useState(null);
  const [tooltipPosition, setTooltipPosition] = useState({ x: 0, y: 0 });
  const messagesContainerRef = useRef(null);
  const navigate = useNavigate();
  
  useEffect(() => {
    fetchStory();
  }, [id]);
  
  useEffect(() => {
    // Scroll to bottom when new messages are displayed
    if (messagesContainerRef.current) {
      messagesContainerRef.current.scrollTop = messagesContainerRef.current.scrollHeight;
    }
  }, [displayedMessages]);
  
  const fetchStory = async () => {
    try {
      setLoading(true);
      const { data } = await axios.get(`${API_URL}/stories/${id}`);
      setStory(data.story);
      
      if (data.userProgress) {
        setUserProgress(data.userProgress);
        setCurrentMessageIndex(data.userProgress.lastMessageIndex);
        
        // Display messages up to the last viewed message
        const messagesToShow = data.story.messages
          .filter(msg => msg.order <= data.userProgress.lastMessageIndex)
          .map(msg => ({
            ...msg,
            isSender: msg.order % 2 === 1 // Alternate sender/receiver for simplicity
          }));
        
        setDisplayedMessages(messagesToShow);
      } else {
        setDisplayedMessages([]);
      }
    } catch (error) {
      console.error('Error fetching story:', error);
      toast.error('Failed to load story');
      navigate('/dashboard');
    } finally {
      setLoading(false);
    }
  };
  
  const showNextMessage = async () => {
    if (!story || !story.messages) return;
    
    // If we've reached the limit for non-premium users, show the modal
    if (story.limitedAccess && currentMessageIndex >= story.messages.length - 1) {
      setShowPremiumModal(true);
      return;
    }
    
    // If we're at the end of available messages
    if (currentMessageIndex >= story.messages.length - 1) {
      // Mark story as completed
      try {
        await axios.post(`${API_URL}/stories/${id}/progress`, {
          messageIndex: currentMessageIndex,
          isCompleted: true
        });
        
        toast.success('Story completed! 🎉');
        return;
      } catch (error) {
        console.error('Error updating progress:', error);
      }
    }
    
    // Show next message
    const nextIndex = currentMessageIndex + 1;
    const nextMessage = story.messages.find(msg => msg.order === nextIndex);
    
    if (nextMessage) {
      // Add to displayed messages
      setDisplayedMessages(prev => [
        ...prev, 
        { ...nextMessage, isSender: nextIndex % 2 === 1 }
      ]);
      
      setCurrentMessageIndex(nextIndex);
      
      // Update progress in database
      try {
        const { data } = await axios.post(`${API_URL}/stories/${id}/progress`, {
          messageIndex: nextIndex,
          isCorrect: false,
          isCompleted: nextIndex === story.messages.length - 1
        });
        
        // Update user XP
        if (data.xpEarned > 0) {
          updateUserXp(data.xpEarned);
          toast.info(`+${data.xpEarned} XP earned!`);
        }
        
        setUserProgress(data.progress);
      } catch (error) {
        console.error('Error updating progress:', error);
      }
    }
  };
  
  const handleQuizAnswer = async (isCorrect) => {
    // Update progress in database with quiz result
    try {
      const { data } = await axios.post(`${API_URL}/stories/${id}/progress`, {
        messageIndex: currentMessageIndex,
        isCorrect,
        isCompleted: currentMessageIndex === story.messages.length - 1
      });
      
      // Update user XP
      if (data.xpEarned > 0) {
        updateUserXp(data.xpEarned);
        toast.info(`+${data.xpEarned} XP earned!`);
      }
      
      setUserProgress(data.progress);
    } catch (error) {
      console.error('Error updating quiz progress:', error);
    }
  };
  
  const handleWordClick = (word, translation, event) => {
    // Get click position for tooltip
    const rect = event.target.getBoundingClientRect();
    
    setSelectedWord({ word, translation });
    setTooltipPosition({
      x: rect.left,
      y: rect.top
    });
  };
  
  // Hide tooltip when clicking elsewhere
  useEffect(() => {
    const handleClickOutside = () => {
      setSelectedWord(null);
    };
    
    document.addEventListener('click', handleClickOutside);
    return () => {
      document.removeEventListener('click', handleClickOutside);
    };
  }, []);
  
  if (loading) {
    return (
      <div className="flex justify-center py-12">
        <div className="animate-spin rounded-full h-12 w-12 border-t-2 border-b-2 border-indigo-600"></div>
      </div>
    );
  }
  
  if (!story) {
    return (
      <div className="text-center py-12">
        <h2 className="text-2xl font-bold text-gray-800 mb-4">Story not found</h2>
        <button 
          onClick={() => navigate('/dashboard')}
          className="btn btn-primary"
        >
          Back to Dashboard
        </button>
      </div>
    );
  }
  
  // Get current message
  const currentMessage = story.messages.find(msg => msg.order === currentMessageIndex);
  
  // Check if current message has a quiz
  const hasQuiz = currentMessage && currentMessage.quizType !== 'none';
  
  return (
    <div className="max-w-3xl mx-auto">
      {/* Story header */}
      <div className="flex items-center mb-6">
        <button 
          onClick={() => navigate('/dashboard')}
          className="mr-4 btn bg-white text-gray-700 hover:bg-gray-100 shadow-sm"
        >
          <FaArrowLeft className="mr-2" /> Back
        </button>
        
        <div>
          <h1 className="text-2xl font-bold text-gray-900">{story.title}</h1>
          <div className="flex items-center mt-1">
            <span className={`px-2 py-1 rounded-full text-xs font-medium ${
              story.level === 'beginner' ? 'bg-green-100 text-green-800' :
              story.level === 'intermediate' ? 'bg-yellow-100 text-yellow-800' :
              'bg-red-100 text-red-800'
            }`}>
              {story.level.charAt(0).toUpperCase() + story.level.slice(1)}
            </span>
            
            {story.tags && story.tags.map((tag, index) => (
              <span key={index} className="ml-2 px-2 py-1 bg-gray-100 text-gray-700 rounded-full text-xs">
                {tag}
              </span>
            ))}
          </div>
        </div>
      </div>
      
      {/* Messages container */}
      <div className="bg-white rounded-t-xl shadow-md p-4 h-96 overflow-y-auto" ref={messagesContainerRef}>
        {displayedMessages.length === 0 ? (
          <div className="h-full flex flex-col items-center justify-center text-gray-500">
            <p className="mb-4">Start the conversation by clicking the button below</p>
            <FaArrowDown size={24} />
          </div>
        ) : (
          <div className="space-y-4">
            {displayedMessages.map((message, index) => (
              <MessageBubble 
                key={index} 
                message={message} 
                onWordClick={(word, translation, e) => handleWordClick(word, translation, e)}
              />
            ))}
          </div>
        )}
        
        {/* Word tooltip */}
        {selectedWord && (
          <WordTooltip 
            word={selectedWord.word}
            translation={selectedWord.translation}
            position={tooltipPosition}
          />
        )}
      </div>
      
      {/* Quiz area */}
      {hasQuiz && (
        <QuizComponent 
          quiz={{
            quizType: currentMessage.quizType,
            question: currentMessage.quizQuestion || "Choose the correct answer:",
            options: currentMessage.options || [],
            correctOption: currentMessage.correctOption
          }}
          onAnswer={handleQuizAnswer}
        />
      )}
      
      {/* Controls */}
      <div className="bg-gray-100 rounded-b-xl shadow-md p-4 flex justify-between items-center">
        <div className="text-sm text-gray-600">
          {story.messages && (
            <>
              Message {currentMessageIndex + 1} of {story.messages.length}
              {userProgress && (
                <span className="ml-4">
                  XP: {userProgress.xpEarned || 0}
                </span>
              )}
            </>
          )}
        </div>
        
        <button
          onClick={showNextMessage}
          className="btn btn-primary flex items-center"
          disabled={loading || (story.limitedAccess && currentMessageIndex >= story.messages.length - 1)}
        >
          {loading ? (
            <FaSpinner className="animate-spin mr-2" />
          ) : (
            <>
              Next <FaArrowRight className="ml-2" />
            </>
          )}
        </button>
      </div>
      
      {/* Premium modal */}
      <PremiumModal 
        isOpen={showPremiumModal} 
        onClose={() => setShowPremiumModal(false)} 
      />
    </div>
  );
};

export default StoryView;
